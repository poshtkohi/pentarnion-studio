cmake_minimum_required(VERSION 3.24)
project(PentarnionAlgebraCheck VERSION 0.2.0 LANGUAGES CXX)
# A small source-only CI project, not a reduced Studio implementation. All
# algorithms and tests are compiled directly from the normal repository files.
set(PENTA_SOURCE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." CACHE PATH "Pentarnion source checkout")
get_filename_component(PENTA_SOURCE_ROOT "${PENTA_SOURCE_ROOT}" ABSOLUTE)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
find_package(Eigen3 CONFIG REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
find_path(PENTA_BOOST_INCLUDE_DIR NAMES boost/multiprecision/cpp_int.hpp REQUIRED)
find_package(Threads REQUIRED)
set(_algebra_sources algebra_context pentarnion elementary universal universal_builtins universal_import universal_hash universal_elementary)
foreach(_name IN LISTS _algebra_sources)
  list(APPEND _sources "${PENTA_SOURCE_ROOT}/src/algebra/${_name}.cpp")
endforeach()
add_library(penta_algebra_check STATIC ${_sources})
target_include_directories(penta_algebra_check PUBLIC "${PENTA_SOURCE_ROOT}/src" PRIVATE "${PENTA_BOOST_INCLUDE_DIR}")
target_link_libraries(penta_algebra_check PRIVATE Eigen3::Eigen nlohmann_json::nlohmann_json)
if(MSVC)
  target_compile_options(penta_algebra_check PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(penta_algebra_check PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
foreach(_suite algebra elementary universal)
  add_executable(penta_${_suite}_tests "${PENTA_SOURCE_ROOT}/tests/algebra/${_suite}_tests.cpp")
  target_link_libraries(penta_${_suite}_tests PRIVATE penta_algebra_check Threads::Threads nlohmann_json::nlohmann_json)
  add_test(NAME ${_suite}_source_check COMMAND penta_${_suite}_tests)
  set_tests_properties(${_suite}_source_check PROPERTIES TIMEOUT 120)
endforeach()
