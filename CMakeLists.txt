  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${_fragment}")

foreach(feature pde_experiments graph_tools networks units)
  if(feature STREQUAL "pde_experiments")
    set(_frontier_source src/modules/pde_experiment_module.cpp)
  elseif(feature STREQUAL "networks")
    set(_frontier_source src/modules/network_module.cpp)
  elseif(feature STREQUAL "units")
    set(_frontier_source src/modules/units_module.cpp)
  else()
    set(_frontier_source src/modules/graph_tools_module.cpp)
  endif()

  target_compile_definitions(penta_${feature} PRIVATE PENTA_NUMERIC_MODULE_BUILD)
  target_include_directories(penta_${feature} PRIVATE src)
  target_link_libraries(penta_${feature} PRIVATE penta_warnings)

    target_link_libraries(penta_${feature} PRIVATE SUNDIALS::cvodes SUNDIALS::nvecserial SUNDIALS::sunlinsolspgmr SUNDIALS::core)
  endif()
  set_target_properties(penta_${feature} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/modules" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/modules")

  else()
    set_target_properties(penta_${feature} PROPERTIES BUILD_RPATH "$ORIGIN/.." INSTALL_RPATH "$ORIGIN/..")
  endif()
  add_dependencies(penta_runtime penta_${feature})
  install(TARGETS penta_${feature} LIBRARY DESTINATION bin/modules RUNTIME DESTINATION bin/modules)
endforeach()
